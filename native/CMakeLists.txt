cmake_minimum_required(VERSION 3.20)
project(hopf_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(Boost CONFIG REQUIRED)
find_program(NODE_EXECUTABLE node REQUIRED)
set(GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_custom_command(OUTPUT "${GENERATED_DIR}/optics.inc"
  COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate-optics.mjs" "${GENERATED_DIR}/optics.inc"
  DEPENDS generate-optics.mjs ../web/shader.js VERBATIM)
add_executable(hopf-render main.cpp "${GENERATED_DIR}/optics.inc")
target_include_directories(hopf-render PRIVATE include "${GENERATED_DIR}" ${Boost_INCLUDE_DIRS})
target_link_libraries(hopf-render PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(hopf-render PRIVATE /O2 /fp:strict)
else()
  target_compile_options(hopf-render PRIVATE -O3 -ffp-contract=off -Wall -Wextra -Wno-unused-parameter)
endif()
find_package(SDL3 QUIET CONFIG)
if(SDL3_FOUND)
  target_link_libraries(hopf-render PRIVATE SDL3::SDL3)
  target_compile_definitions(hopf-render PRIVATE HOPF_SDL=1)
endif()
